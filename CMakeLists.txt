cmake_minimum_required(VERSION 3.20)
project(CPPNN VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(mnist 1mnist/main.cpp)
add_executable(cifar10 2CIFAR10/main.cpp)
option(CPPNN_BUILD_GUI "Build the optional OpenCV digit UI" OFF)
if(CPPNN_BUILD_GUI)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui)
    add_executable(digit_ui 1mnist/digit_ui.cpp)
    target_include_directories(digit_ui PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(digit_ui PRIVATE ${OpenCV_LIBS})
endif()
include(CTest)
if(BUILD_TESTING)
    foreach(test data mnist cifar)
        add_executable(test_${test} tests/test_${test}.cpp)
        add_test(NAME ${test} COMMAND test_${test})
    endforeach()
    add_test(NAME mnist_help COMMAND mnist --help)
    add_test(NAME cifar_help COMMAND cifar10 --help)
    add_test(NAME mnist_missing_data COMMAND mnist --data missing-dataset --limit 1)
    add_test(NAME cifar_missing_data COMMAND cifar10 --data missing-dataset --limit 1)
    set_tests_properties(mnist_missing_data cifar_missing_data PROPERTIES WILL_FAIL TRUE)
endif()
